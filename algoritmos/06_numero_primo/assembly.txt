JUMP main
primo:
LOAD_STACK $at
STORE $at, n, 0
STORE_STACK $ra
LOAD $t0, n
ADDI $t1, $zero, 2
BGE $t0, $t1, L0
ADDI $t0, $zero, 0
LOAD_STACK $ra
STORE_STACK $t0
JR $ra
JUMP L1
L0:
L1:
ADDI $t0, $zero, 2
STORE -, $t0, divisor
L2:
LOAD $t0, divisor
LOAD $t1, divisor
MULT $t2, $t0, $t1
LOAD $t0, n
BGT $t2, $t0, L3
LOAD $t0, n
LOAD $t1, n
LOAD $t2, divisor
DIV $t3, $t1, $t2
LOAD $t1, divisor
MULT $t2, $t3, $t1
SUB $t1, $t0, $t2
STORE -, $t1, resto
LOAD $t0, resto
ADDI $t1, $zero, 0
BNE $t0, $t1, L4
ADDI $t0, $zero, 0
LOAD_STACK $ra
STORE_STACK $t0
JR $ra
JUMP L5
L4:
L5:
LOAD $t0, divisor
ADDI $t1, $zero, 1
ADD $t2, $t0, $t1
STORE -, $t2, divisor
JUMP L2
L3:
ADDI $t0, $zero, 1
LOAD_STACK $ra
STORE_STACK $t0
JR $ra
LOAD_STACK $ra
JR $ra
main:
STORE_STACK $ra
IN $t0
STORE -, $t0, n
LOAD $at, n, 0
STORE_STACK $at
LOAD $t0, n
STORE_STACK $t0
JAL $ra, primo
LOAD_STACK $t0
LOAD_STACK $at
STORE $at, n, 0
OUT $t0
HALT -

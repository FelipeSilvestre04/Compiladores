void multiplica(int a[], int b[], int c[])
{
    int i;
    int j;
    int k;
    int total;

    i = 0;
    while (i < 2)
    {
        j = 0;
        while (j < 2)
        {
            total = 0;
            k = 0;
            while (k < 2)
            {
                total = total + a[i * 2 + k] * b[k * 2 + j];
                k = k + 1;
            }
            c[i * 2 + j] = total;
            j = j + 1;
        }
        i = i + 1;
    }
}

void main(void)
{
    int a[4];
    int b[4];
    int c[4];
    int i;

    i = 0;
    while (i < 4)
    {
        a[i] = input();
        i = i + 1;
    }

    i = 0;
    while (i < 4)
    {
        b[i] = input();
        i = i + 1;
    }

    multiplica(a, b, c);

    i = 0;
    while (i < 4)
    {
        output(c[i]);
        i = i + 1;
    }
}

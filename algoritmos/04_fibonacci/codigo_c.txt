int fibonacci(int n)
{
    int a;
    int b;
    int i;
    int proximo;

    if (n <= 0)
        return 0;

    a = 0;
    b = 1;
    i = 1;

    while (i < n)
    {
        proximo = a + b;
        a = b;
        b = proximo;
        i = i + 1;
    }

    return b;
}

void main(void)
{
    int n;
    n = input();
    output(fibonacci(n));
}
